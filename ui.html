API documentation

Base URL — every request needs the API key as X-API-Key header or ?key= query parameter.

Send a message

POST /send
{"to": "60123456789", "text": "hello"}

POST /send   (image by URL, optional caption)
{"to": "60123456789", "imageUrl": "https://example.com/pic.jpg", "caption": "look"}

"to" accepts a bare number or a group id like 1203630...@g.us. Returns {"ok":true,"id":"...","to":"..."}.

Read messages

GET /messages?limit=100
  after=<seq>               only messages after this cursor (for polling)
  chat=<number|group id>    one conversation
  from=<number>             one sender
  q=<text>                  full-text search
  order=desc                newest first (default oldest first)
  limit=1..500              default 100

Each message: {seq, id, from, chat, isGroup, fromMe, name, type, text, timestamp, hasMedia}. from is a plain number ("me" for your own messages). Poll by repeating with after=<last cursor>.

Chats

GET /chats

One row per conversation, newest first, with resolved names and the last message.

Media & pictures

GET /media/<message id>              image / voice / video / document file
GET /avatar/<number or group id>     profile picture (SVG placeholder if none)

Media downloads from WhatsApp on first request and is cached on disk afterwards.

Connection

GET  /status      state, account, sync progress, database stats
GET  /qr          QR page for linking (open in browser)
GET  /qr.json     QR as JSON {state, user, qr: dataUrl}
POST /logout      unlink; body {"clearData": true} also wipes the local database
POST /sync        re-run group name / avatar / media sync

Settings

GET  /settings
POST /settings    {"webhookUrl": "...", "keepDays": 90, "mediaSyncDays": 30, "port": 3000}

Message format

Every message — from GET /messages and from the webhook — has this exact shape:

{
  "seq": 10412,                     // database cursor (not sent to webhooks)
  "id": "3EB0F60D6CE9DBEAB93486",   // WhatsApp message id, use it for /media/<id>
  "from": "60123456789",            // sender's phone number, or "me" if you sent it
  "chat": "60123456789",            // conversation: a number, or "1203...@g.us" for groups
  "isGroup": false,
  "fromMe": false,
  "name": "John Tan",               // sender's name (contact book / display name)
  "type": "conversation",           // conversation | extendedTextMessage | imageMessage |
                                    // videoMessage | audioMessage | documentMessage | stickerMessage
  "text": "hello",                  // message text, caption, or document filename
  "timestamp": 1784708117000,       // milliseconds — divide by 1000 for PHP date()
  "hasMedia": false                 // true = a file is downloadable at /media/<id>
}

Server states from GET /status: connected (ready to send), waiting_for_qr_scan (needs linking), connecting, reconnecting, logged_out.

Errors: 401 wrong/missing API key · 400 bad input (e.g. invalid "to") · 503 WhatsApp not connected · 502 media download failed. Error body is always {"error": "reason"}.

PHP — the helper (use it for every call)

<?php
// wa.php — include this once, then every API call is one line
define('WA_URL', 'http://localhost:3210');   // installed app port; dev server uses 3000
define('WA_KEY', 'YOUR_API_KEY');            // Settings -> Integration -> API key

function wa($method, $path, $data = null) {
    $ch = curl_init(WA_URL . $path);
    $headers = ['X-API-Key: ' . WA_KEY];
    if ($data !== null) {
        curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode($data));
        $headers[] = 'Content-Type: application/json';
    }
    curl_setopt_array($ch, [
        CURLOPT_CUSTOMREQUEST  => $method,
        CURLOPT_HTTPHEADER     => $headers,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_TIMEOUT        => 60,
    ]);
    $res = curl_exec($ch);
    if ($res === false) throw new Exception('WhatsApp Server unreachable: ' . curl_error($ch));
    $code = curl_getinfo($ch, CURLINFO_HTTP_CODE);
    $json = json_decode($res, true);
    if ($code >= 400) throw new Exception($json['error'] ?? ('HTTP ' . $code));
    return $json;
}

PHP — check server status before sending

$status = wa('GET', '/status');

if ($status['state'] !== 'connected') {
    die('WhatsApp is not ready: ' . $status['state']);
}
echo 'Connected as +' . $status['user']['number'];
echo ' — ' . $status['stats']['messages'] . ' messages stored';

PHP — send messages

// plain text to a number
$r = wa('POST', '/send', ['to' => '60123456789', 'text' => 'Hello from PHP!']);
echo 'sent, id = ' . $r['id'];

// image from a URL, with caption
wa('POST', '/send', [
    'to'       => '60123456789',
    'imageUrl' => 'https://example.com/receipt.jpg',
    'caption'  => 'Your receipt',
]);

// to a group (use the chat id from /chats or /messages)
wa('POST', '/send', ['to' => '120363405006268972@g.us', 'text' => 'Hi group']);

PHP — read, search and poll messages

// last 50 messages of one conversation, newest first
$r = wa('GET', '/messages?chat=60123456789&order=desc&limit=50');
foreach ($r['messages'] as $m) {
    echo date('Y-m-d H:i:s', intval($m['timestamp'] / 1000))
       . ($m['fromMe'] ? ' me: ' : ' ' . $m['name'] . ': ') . $m['text'] . PHP_EOL;
}

// search all history for a word
$r = wa('GET', '/messages?q=' . urlencode('invoice') . '&limit=100');

// poll for NEW messages (e.g. in a cron job) — keep the cursor between runs
$cursor = intval(@file_get_contents('cursor.txt'));
$r = wa('GET', '/messages?after=' . $cursor);
foreach ($r['messages'] as $m) {
    if (!$m['fromMe']) {
        // handle incoming message here
    }
}
file_put_contents('cursor.txt', $r['cursor']);

PHP — receive messages by webhook (recommended)

Set your URL in Settings → Integration (e.g. http://localhost/my_app/receive_whatsapp.php). The server POSTs each incoming message to it instantly — no polling.

<?php
// receive_whatsapp.php
require 'wa.php';

$msg = json_decode(file_get_contents('php://input'), true);
if (!$msg || empty($msg['id'])) { http_response_code(400); exit; }

// 1. log it
file_put_contents('whatsapp.log',
    date('c') . " {$msg['from']} ({$msg['name']}): {$msg['text']}\n", FILE_APPEND);

// 2. auto-reply example
if (strtolower(trim($msg['text'])) === 'price') {
    wa('POST', '/send', ['to' => $msg['chat'], 'text' => 'Our price list: ...']);
}

// 3. save attached media (photo / voice / document)
if (!empty($msg['hasMedia'])) {
    $file = file_get_contents(WA_URL . '/media/' . $msg['id'] . '?key=' . WA_KEY);
    if ($file !== false) file_put_contents('uploads/' . $msg['id'], $file);
}

http_response_code(200); // always answer 200 quickly

PHP — download media & profile pictures

// media file for any message where hasMedia = true
$jpg = file_get_contents(WA_URL . '/media/3EB0F60D6CE9DBEAB93486?key=' . WA_KEY);
file_put_contents('photo.jpg', $jpg);

// someone's profile picture
$avatar = file_get_contents(WA_URL . '/avatar/60123456789?key=' . WA_KEY);

// list all conversations
$chats = wa('GET', '/chats')['chats'];
foreach ($chats as $c) {
    echo ($c['isGroup'] ? '[group] ' : '') . $c['name'] . ' — ' . $c['lastText'] . PHP_EOL;
}

PHP — manage the server

// trigger a data re-sync (group names, pictures, media)
wa('POST', '/sync');

// read or change settings
$s = wa('GET', '/settings');
wa('POST', '/settings', ['webhookUrl' => 'http://localhost/my_app/receive_whatsapp.php']);

// unlink the WhatsApp account (clearData true also wipes the local database)
wa('POST', '/logout', ['clearData' => false]);
